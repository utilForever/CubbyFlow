93template <size_t N>
94unsigned int MPMFluidSolver<N>::GetNumberOfSubTimeSteps(
95 double timeIntervalInSeconds) const
96{
97 if (!std::isfinite(timeIntervalInSeconds) || timeIntervalInSeconds <= 0.0 ||
98 m_mpmSystemData->NumberOfParticles() == 0)
99 {
100 return 1;
101 }
102
103 const auto spacing = m_mpmSystemData->GridMass().GridSpacing();
104 const auto velocities = m_mpmSystemData->Velocities();
105 double minSpacing = spacing[0];
106 double maxSpeed = 0.0;
107
108 for (size_t axis = 1; axis < N; ++axis)
109 {
110 minSpacing = std::min(minSpacing, spacing[axis]);
111 }
112
113 for (const auto& velocity : velocities)
114 {
115 for (double component : velocity)
116 {
117 if (!std::isfinite(component))
118 {
119 throw std::invalid_argument{
120 "Invalid MPM fluid particle velocity."
121 };
122 }
123 }
124
125 const double speed = velocity.Length();
126
127 if (!std::isfinite(speed))
128 {
129 throw std::invalid_argument{
130 "Invalid MPM fluid particle velocity."
131 };
132 }
133
134 maxSpeed = std::max(maxSpeed, speed);
135 }
136
137 const double desiredTimeStep =
138 minSpacing / (m_constitutiveModel.GetSpeedOfSound() + maxSpeed);
139 const double count = std::ceil(timeIntervalInSeconds /
140 (m_timeStepLimitScale * desiredTimeStep));
141 const double maxCount =
142 static_cast<double>(std::numeric_limits<unsigned int>::max());
143
144 return static_cast<unsigned int>(std::clamp(count, 1.0, maxCount));
145}

165{

177template <size_t N>
178void MPMFluidSolver<N>::InitializeReferenceVolumes()
179{
180 const auto masses = m_mpmSystemData->ParticleMasses();
181 auto volumes = m_mpmSystemData->InitialVolumes();
182
183 for (size_t i = 0; i < volumes.Length(); ++i)
184 {
185 if (volumes[i] == 0.0)
186 {
187 volumes[i] = masses[i] / m_constitutiveModel.GetTargetDensity();
188 }
189 else if (!std::isfinite(volumes[i]) || volumes[i] < 0.0)
190 {
191 throw std::invalid_argument{
192 "Invalid MPM fluid reference volume."
193 };
194 }
195 }

198template <size_t N>
199void MPMFluidSolver<N>::UpdateGridVelocities(double timeStepInSeconds)
200{
201 const auto positions = m_mpmSystemData->Positions();
202 const auto velocities = m_mpmSystemData->Velocities();
203 const auto masses = m_mpmSystemData->ParticleMasses();
204 const auto initialVolumes = m_mpmSystemData->InitialVolumes();
205 const auto volumeRatios = m_mpmSystemData->VolumeRatios();
206 const auto& gridMass = m_mpmSystemData->GridMass();
207 auto& gridVelocities = m_mpmSystemData->GridVelocities();
208 const auto dataSize = gridMass.DataSize();
209 const auto spacing = gridMass.GridSpacing();
210 const auto dataOrigin = gridMass.DataOrigin();
211 const auto isFinite = [](double value) { return std::isfinite(value); };
212
213 for (size_t i = 0; i < positions.Length(); ++i)
214 {
215 const double currentVolume = initialVolumes[i] * volumeRatios[i];
216 const double density =
217 m_constitutiveModel.ComputeDensity(masses[i], currentVolume);
218 const double pressure = m_constitutiveModel.ComputePressure(density);
219 const MatrixType stress =
220 m_constitutiveModel.ComputeCauchyStress(pressure);
221 const VectorType relativeVelocity =
222 velocities[i] - this->GetWind()->Sample(positions[i]);
223 const VectorType externalForce =
224 masses[i] * this->GetGravity() -
225 this->GetDragCoefficient() * relativeVelocity;
226 const auto stencil = CubicBSplineKernel<N>::GetStencil(
227 positions[i], spacing, dataOrigin);
228
229 for (const auto& entry : stencil)
230 {
231 const auto index = ClampIndex(entry.index, dataSize);
232 const double nodeMass = gridMass(index);
233
234 if (nodeMass <= 0.0)
235 {
236 continue;
237 }
238
239 const VectorType increment =
240 timeStepInSeconds *
241 (entry.weight * externalForce -
242 currentVolume * stress * entry.gradient) /
243 nodeMass;
244
245 if (!std::ranges::all_of(increment, isFinite))
246 {
247 throw std::invalid_argument{ "Non-finite MPM fluid update." };
248 }
249
250 gridVelocities(index) += increment;
251 }
252 }
