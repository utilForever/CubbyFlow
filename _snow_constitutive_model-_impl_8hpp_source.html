23template <size_t N>
24SnowConstitutiveModel<N>::SnowConstitutiveModel(double youngsModulus,
25 double poissonRatio,
26 double criticalCompression,
27 double criticalStretch,
28 double hardeningCoefficient)
29 : m_criticalCompression(criticalCompression),
30 m_criticalStretch(criticalStretch),
31 m_hardeningCoefficient(hardeningCoefficient)
32{
33 if (const std::array parameterChecks = { std::isfinite(youngsModulus),
34 youngsModulus > 0.0,
35 std::isfinite(poissonRatio),
36 poissonRatio > -1.0,
37 poissonRatio < 0.5,
38 std::isfinite(criticalCompression),
39 criticalCompression >= 0.0,
40 criticalCompression < 1.0,
41 std::isfinite(criticalStretch),
42 criticalStretch >= 0.0,
43 std::isfinite(
44 hardeningCoefficient),
45 hardeningCoefficient >= 0.0 };
46 !std::ranges::all_of(parameterChecks,
47 [](bool isValid) { return isValid; }))
48 {
49 throw std::invalid_argument{ "Invalid snow material parameters." };
50 }
51
52 m_mu0 = youngsModulus / (2.0 * (1.0 + poissonRatio));
53 m_lambda0 = youngsModulus * poissonRatio /
54 ((1.0 + poissonRatio) * (1.0 - 2.0 * poissonRatio));
55
56 if (!std::isfinite(m_mu0) || !std::isfinite(m_lambda0))
57 {
58 throw std::invalid_argument{ "Invalid snow material parameters." };
59 }
60}

62template <size_t N>
63typename SnowConstitutiveModel<N>::State SnowConstitutiveModel<N>::Update(
64 const MatrixType& deformationGradientIncrement, const State& state) const
65{
66 ValidateDeformation(deformationGradientIncrement);
67 ValidateDeformation(state.elastic);
68 ValidateDeformation(state.plastic);
69
70 const MatrixType trialElastic =
71 deformationGradientIncrement * state.elastic;
72 const MatrixType trialTotal = trialElastic * state.plastic;
73
74 ValidateDeformation(trialElastic);
75 ValidateDeformation(trialTotal);
76
77 MatrixType u;
78 Vector<double, N> singularValues;
79 MatrixType v;
80
81 SVD(trialElastic, u, singularValues, v);
82
83 for (size_t i = 0; i < N; ++i)
84 {
85 singularValues[i] =
86 std::clamp(singularValues[i], 1.0 - m_criticalCompression,
87 1.0 + m_criticalStretch);
88 }
89
90 State result;
91 result.elastic =
92 u * MatrixType::MakeScaleMatrix(singularValues) * v.Transposed();
93 result.plastic = result.elastic.Inverse() * trialTotal;
94
95 ValidateDeformation(result.elastic);
96 ValidateDeformation(result.plastic);
97
98 return result;
99}

101template <size_t N>
102typename SnowConstitutiveModel<N>::MatrixType
103SnowConstitutiveModel<N>::ComputeKirchhoffStress(const State& state) const
104{
105 ValidateDeformation(state.elastic);
106 ValidateDeformation(state.plastic);
107
108 MatrixType u;
109 Vector<double, N> singularValues;
110 MatrixType v;
111
112 SVD(state.elastic, u, singularValues, v);
113
114 const MatrixType rotation = u * v.Transposed();
115 const double elasticDeterminant = state.elastic.Determinant();
116 const double hardening = ComputeHardening(state);
117
118 const double mu = m_mu0 * hardening;
119 const double lambda = m_lambda0 * hardening;
120 const MatrixType stress =
121 2.0 * mu * (state.elastic - rotation) * state.elastic.Transposed() +
122 lambda * (elasticDeterminant - 1.0) * elasticDeterminant *
123 MatrixType::MakeIdentity();
124
125 if (!IsFinite(stress))
126 {
127 throw std::invalid_argument{ "Non-finite snow stress." };
128 }
129
130 return stress;
131}

133template <size_t N>
134double SnowConstitutiveModel<N>::ComputeWaveSpeed(const State& state,
135 double referenceDensity) const
136{
137 ValidateDeformation(state.elastic);
138 ValidateDeformation(state.plastic);
139
140 if (!std::isfinite(referenceDensity) || referenceDensity <= 0.0)
141 {
142 throw std::invalid_argument{ "Invalid snow reference density." };
143 }
144
145 MatrixType u;
146 Vector<double, N> singularValues;
147 MatrixType v;
148 SVD(state.elastic, u, singularValues, v);
149
150 const double elasticDeterminant = state.elastic.Determinant();
151 const double totalDeterminant =
152 elasticDeterminant * state.plastic.Determinant();
153 const double currentDensity = referenceDensity / totalDeterminant;
154 const double hardening = ComputeHardening(state);
155 const double mu = m_mu0 * hardening;
156 const double lambda = m_lambda0 * hardening;
157 double maxCandidate = 0.0;
158
159 for (size_t a = 0; a < N; ++a)
160 {
161 const double sigmaA = singularValues[a];
162 maxCandidate = std::max(
163 maxCandidate, 2.0 * mu * sigmaA * sigmaA +
164 lambda * elasticDeterminant * elasticDeterminant);
165
166 for (size_t b = 0; b < N; ++b)
167 {
168 if (a == b)
169 {
170 continue;
171 }
172
173 const double sigmaB = singularValues[b];
174 maxCandidate = std::max(maxCandidate, 2.0 * mu * sigmaB * sigmaB *
175 (sigmaA + sigmaB - 1.0) /
176 (sigmaA + sigmaB));
177 }
178 }
179
180 const double kappa = maxCandidate / totalDeterminant;
181 const double waveSpeed = std::sqrt(kappa / currentDensity);
182
183 if (!std::isfinite(totalDeterminant) || totalDeterminant <= 0.0 ||
184 !std::isfinite(currentDensity) || currentDensity <= 0.0 ||
185 !std::isfinite(kappa) || kappa <= 0.0 || !std::isfinite(waveSpeed))
186 {
187 throw std::invalid_argument{ "Invalid snow wave speed." };
188 }
189
190 return waveSpeed;
191}

193template <size_t N>
194double SnowConstitutiveModel<N>::ComputeHardening(const State& state) const
195{
196 const double hardening =
197 std::exp(m_hardeningCoefficient * (1.0 - state.plastic.Determinant()));
198
199 if (!std::isfinite(hardening))
200 {
201 throw std::invalid_argument{ "Non-finite snow hardening." };
202 }
203
204 return hardening;
205}
