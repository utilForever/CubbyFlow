82template <size_t N>
83unsigned int SnowMPMSolver<N>::GetNumberOfSubTimeSteps(
84 double timeIntervalInSeconds) const
85{
86 if (!std::isfinite(timeIntervalInSeconds) || timeIntervalInSeconds <= 0.0)
87 {
88 return 1;
89 }
90
91 const auto velocities = m_mpmSystemData->Velocities();
92 const auto masses = m_mpmSystemData->ParticleMasses();
93 const auto volumes = m_mpmSystemData->InitialVolumes();
94 const auto states = m_mpmSystemData->DeformationStates();
95 const auto spacing = m_mpmSystemData->GridMass().GridSpacing();
96 double minSpacing = spacing[0];
97 double maxVelocity = 0.0;
98 double maxWaveSpeed = 0.0;
99
100 for (size_t axis = 1; axis < N; ++axis)
101 {
102 minSpacing = std::min(minSpacing, spacing[axis]);
103 }
104
105 for (size_t i = 0; i < velocities.Length(); ++i)
106 {
107 maxVelocity = std::max(maxVelocity, velocities[i].Length());
108
109 if (volumes[i] == 0.0)
110 {
111 continue;
112 }
113
114 if (!std::isfinite(volumes[i]) || volumes[i] < 0.0)
115 {
116 throw std::invalid_argument{ "Invalid snow reference volume." };
117 }
118
119 maxWaveSpeed =
120 std::max(maxWaveSpeed, m_constitutiveModel.ComputeWaveSpeed(
121 states[i], masses[i] / volumes[i]));
122 }
123
124 double desiredTimeStep = std::numeric_limits<double>::infinity();
125 if (maxVelocity > 0.0)
126 {
127 desiredTimeStep = std::min(desiredTimeStep, minSpacing / maxVelocity);
128 }
129 if (maxWaveSpeed > 0.0)
130 {
131 desiredTimeStep = std::min(desiredTimeStep, minSpacing / maxWaveSpeed);
132 }
133 if (m_maxVelocityGradient > 0.0)
134 {
135 desiredTimeStep =
136 std::min(desiredTimeStep, 0.2 / m_maxVelocityGradient);
137 }
138
139 if (!std::isfinite(desiredTimeStep))
140 {
141 return 1;
142 }
143
144 const double count = std::ceil(timeIntervalInSeconds /
145 (m_timeStepLimitScale * desiredTimeStep));
146 const double maxCount =
147 static_cast<double>(std::numeric_limits<unsigned int>::max());
148
149 return static_cast<unsigned int>(std::clamp(count, 1.0, maxCount));
150}

183template <size_t N>
184void SnowMPMSolver<N>::InitializeReferenceVolumes()
185{
186 const auto positions = m_mpmSystemData->Positions();
187 const auto masses = m_mpmSystemData->ParticleMasses();
188 auto volumes = m_mpmSystemData->InitialVolumes();
189 const auto& gridMass = m_mpmSystemData->GridMass();
190 const auto dataSize = gridMass.DataSize();
191 const auto spacing = gridMass.GridSpacing();
192 const auto dataOrigin = gridMass.DataOrigin();
193 double cellVolume = 1.0;
194
195 for (size_t axis = 0; axis < N; ++axis)
196 {
197 cellVolume *= spacing[axis];
198 }
199
200 if (!std::isfinite(cellVolume) || cellVolume <= 0.0)
201 {
202 throw std::invalid_argument{ "Invalid snow MPM cell volume." };
203 }
204
205 for (size_t i = 0; i < volumes.Length(); ++i)
206 {
207 if (volumes[i] != 0.0)
208 {
209 if (!std::isfinite(volumes[i]) || volumes[i] < 0.0)
210 {
211 throw std::invalid_argument{ "Invalid snow reference volume." };
212 }
213 continue;
214 }
215
216 double referenceDensity = 0.0;
217 const auto stencil = CubicBSplineKernel<N>::GetStencil(
218 positions[i], spacing, dataOrigin);
219
220 for (const auto& entry : stencil)
221 {
222 referenceDensity += gridMass(ClampIndex(entry.index, dataSize)) *
223 entry.weight / cellVolume;
224 }
225
226 if (!std::isfinite(referenceDensity) || referenceDensity <= 0.0)
227 {
228 throw std::invalid_argument{ "Invalid snow reference density." };
229 }
230
231 volumes[i] = masses[i] / referenceDensity;
232 }
233}

235template <size_t N>
236void SnowMPMSolver<N>::UpdateGridVelocities(double timeStepInSeconds)
237{
238 const auto positions = m_mpmSystemData->Positions();
239 const auto velocities = m_mpmSystemData->Velocities();
240 const auto masses = m_mpmSystemData->ParticleMasses();
241 const auto volumes = m_mpmSystemData->InitialVolumes();
242 const auto states = m_mpmSystemData->DeformationStates();
243 const auto& gridMass = m_mpmSystemData->GridMass();
244 auto& gridVelocities = m_mpmSystemData->GridVelocities();
245 const auto dataSize = gridMass.DataSize();
246 const auto spacing = gridMass.GridSpacing();
247 const auto dataOrigin = gridMass.DataOrigin();
248
249 for (size_t i = 0; i < positions.Length(); ++i)
250 {
251 const VectorType relativeVelocity =
252 velocities[i] - this->GetWind()->Sample(positions[i]);
253 const VectorType externalForce =
254 masses[i] * this->GetGravity() -
255 this->GetDragCoefficient() * relativeVelocity;
256 const MatrixType stress =
257 m_constitutiveModel.ComputeKirchhoffStress(states[i]);
258 const auto stencil = CubicBSplineKernel<N>::GetStencil(
259 positions[i], spacing, dataOrigin);
260
261 for (const auto& entry : stencil)
262 {
263 if (entry.weight == 0.0)
264 {
265 continue;
266 }
267
268 const auto index = ClampIndex(entry.index, dataSize);
269 const double nodeMass = gridMass(index);
270 if (nodeMass > 0.0)
271 {
272 gridVelocities(index) +=
273 timeStepInSeconds *
274 (entry.weight * externalForce -
275 volumes[i] * stress * entry.gradient) /
276 nodeMass;
277 }
278 }
279 }

282template <size_t N>
283typename SnowMPMSolver<N>::MatrixType SnowMPMSolver<N>::ComputeVelocityGradient(
284 size_t particleIndex) const
285{
286 const auto positions = m_mpmSystemData->Positions();
287 const auto& gridVelocities = m_mpmSystemData->GridVelocities();
288 const auto dataSize = gridVelocities.DataSize();
289 const auto stencil = CubicBSplineKernel<N>::GetStencil(
290 positions[particleIndex], gridVelocities.GridSpacing(),
291 gridVelocities.DataOrigin());
292 MatrixType result;
293
294 for (const auto& entry : stencil)
295 {
296 if (entry.weight != 0.0)
297 {
298 const VectorType velocity =
299 gridVelocities(ClampIndex(entry.index, dataSize));
300 for (size_t row = 0; row < N; ++row)
301 {
302 for (size_t column = 0; column < N; ++column)
303 {
304 result(row, column) +=
305 velocity[row] * entry.gradient[column];
306 }
307 }
308 }
309 }
310
311 return result;
312}

314template <size_t N>
315double SnowMPMSolver<N>::ComputeMaxVelocityGradient() const
316{
317 double result = 0.0;
318
319 for (size_t i = 0; i < m_mpmSystemData->NumberOfParticles(); ++i)
320 {
321 result = std::max(result, ComputeVelocityGradient(i).AbsMax());
322 }
323
324 return result;
325}

327template <size_t N>
328void SnowMPMSolver<N>::UpdateDeformation(double timeStepInSeconds)
329{
330 auto states = m_mpmSystemData->DeformationStates();
331 m_maxVelocityGradient = 0.0;
332
333 for (size_t i = 0; i < states.Length(); ++i)
334 {
335 const MatrixType velocityGradient = ComputeVelocityGradient(i);
336 m_maxVelocityGradient =
337 std::max(m_maxVelocityGradient, velocityGradient.AbsMax());
338 states[i] = m_constitutiveModel.Update(
339 MatrixType::MakeIdentity() + timeStepInSeconds * velocityGradient,
340 states[i]);
341 }
342}
